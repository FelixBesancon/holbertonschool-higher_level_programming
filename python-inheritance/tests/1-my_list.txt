# Basic example (must not modify original list)
>>> my_list = MyList([1, 4, 2, 3, 5])
>>> my_list
[1, 4, 2, 3, 5]
>>> my_list.print_sorted()
[1, 2, 3, 4, 5]
>>> print(my_list)
[1, 4, 2, 3, 5]

# Already sorted
>>> sorted_list = MyList([1, 2, 3])
>>> sorted_list.print_sorted()
[1, 2, 3]
>>> print(sorted_list)
[1, 2, 3]

# Reverse sorted
>>> rev = MyList([5, 4, 3, 2, 1])
>>> rev.print_sorted()
[1, 2, 3, 4, 5]
>>> print(rev)
[5, 4, 3, 2, 1]

# Duplicates
>>> dup = MyList([3, 1, 2, 3, 1])
>>> dup.print_sorted()

# Negatives

# Other element types (outside the project assumption: "all int")

# Floats
>>> floats = MyList([3.14, 2.71, -1.0, 0.0])
>>> floats.print_sorted()
[-1.0, 0.0, 2.71, 3.14]
>>> floats
[3.14, 2.71, -1.0, 0.0]

# Bools (comparable: False < True)
>>> bools = MyList([True, False, True, False])
>>> bools.print_sorted()
[False, False, True, True]
>>> bools
[True, False, True, False]

# Lists of lists (lexicographic comparison works in Python)
>>> list_of_lists = MyList([[2, 0], [1, 9], [1, 2], [2]])
>>> list_of_lists.print_sorted()
[[1, 2], [1, 9], [2], [2, 0]]
>>> list_of_lists
[[2, 0], [1, 9], [1, 2], [2]]

# Tuples (lexicographic comparison works)
>>> tuples = MyList([(2, 0), (1, 9), (1, 2), (2,)])
>>> tuples.print_sorted()
[(1, 2), (1, 9), (2,), (2, 0)]
>>> tuples
[(2, 0), (1, 9), (1, 2), (2,)]

# Sets can be sorted (subset ordering)
>>> sets = MyList([{1, 2}, {1}, {1, 2, 3}])
>>> sets.print_sorted()
[{1}, {1, 2}, {1, 2, 3}]
>>> sets
[{1, 2}, {1}, {1, 2, 3}]
